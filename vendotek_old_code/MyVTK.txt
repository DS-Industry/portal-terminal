	start	:BOOL;	//Старт обмена с Вендотеком
	step	:INT;	//Шаги выполнения кейса

	Send_IDL, Send_VRP, Send_FIN, Send_ABR	:BOOL;
	
	ParamId		:BYTE;	//идентификатор принимаемого параметра

	ResvData:ARRAY[1..50] OF BYTE;	//Данные полученные от Вендотек
		
	sum1,sum2,sum3,sum4: BYTE; //Дробим входную сумму на отдельные символы
	CHRsum1,CHRsum2,CHRsum3,CHRsum4: BYTE; //Для преобразований входной суммы
	STRsum1,STRsum2,STRsum3,STRsum4: STRING; //Для преобразований входной суммы

	opros:INT; //Переменная опроса

	r_trig_SUM_IN	:r_trig;
	r_trig_Cancel	:r_trig;
	SendMoney	:BOOL;
	SendMoneyStep	:INT;
	resetSumOut	:ton;
	tonShowScreenPay	:ton;	//Задержка показа экрана приложите карту, т.к. сумма на Вендотек отображается не сразу
	StartTonShowScreenPay	:BOOL;
END_VAR


//Функция BYTE_TO_STRH конвертирует значение IN типа BYTE в строку, содержащую это значение в 16-ричной системе счисления.

resetSumOut(IN:= , PT:= T#1S, Q=> , ET=> );

r_trig_SUM_IN(CLK:= SUM_IN>0, Q=> );
IF r_trig_SUM_IN.Q THEN
	SendMoney := TRUE;
	SendMoneyStep := 1;
END_IF

r_trig_Cancel(CLK:= Cancel, Q=> );
IF r_trig_Cancel.Q THEN
	Send_ABR := TRUE;
	SendMoney := FALSE;
END_IF

IF SendMoney THEN
	CASE SendMoneyStep OF		
	1:	
		Send_IDL := TRUE;
		ShowScreenPay := FALSE;
		HideScreenPay := FALSE;
		SendMoneyStep:=2;
		
	2:	
		IF Send_IDL = FALSE THEN
			Send_VRP := TRUE;
			SendMoneyStep:=3;
			StartTonShowScreenPay := TRUE;
		END_IF
	
	3:	
		IF Send_VRP = FALSE THEN	//Закончился обмен VRP
			IF LEN(STR:= ApprovedAmount)>=3 THEN
				SUM_OUT:=STRING_TO_WORD(LEFT(STR:= ApprovedAmount, SIZE:= LEN(STR:= ApprovedAmount)-2));
			ELSE
				SUM_OUT:=STRING_TO_WORD(ApprovedAmount);
			END_IF
			//SUM_IN := 0;
			resetSumOut.IN := TRUE;	//активируем таймер на обнуление.
			
			IF SUM_OUT > 0 THEN		//получили сумму, шлем FIN
				Send_FIN := TRUE;
				SendMoneyStep:=4;
			ELSE
				SendMoneyStep:=5;
			END_IF
			
			HideScreenPay := TRUE;	//Снимать будем после скрытия
		END_IF		
	
	4:	
		IF Send_FIN = FALSE THEN
			SendMoneyStep:=5;
		END_IF
	
	
	5:	//успешно завершили
		SUM_IN := 0;
		SendMoney := FALSE;	
	END_CASE
END_IF

	resetSumOut.IN := FALSE;

	StartTonShowScreenPay := FALSE;
	ShowScreenPay := TRUE;
END_IF
